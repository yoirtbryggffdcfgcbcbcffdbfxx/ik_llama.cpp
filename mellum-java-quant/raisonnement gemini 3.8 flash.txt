async _download(url, fileName) {
    const requestOptions = {
        method: 'GET',
        mode: 'no-cors',
        referrerPolicy: 'no-referrer'
    }

    fetch(url, requestOptions)
        .then(async (result) => {
            if (!result.ok) throw getTranslation('error_while_updating_downloading_data_file_x', { data_file_name: fileName });
            return result.blob();
        })
        .then(blob => { this._downloadBlob(blob, fileName); })
        .catch(err => { DevExpress.ui.notify(err, "error", 3000) });
}

async _downloadBlob(blob, fileName) {
    const $a = document.createElement('a');
    if (fileName) {
        $a.setAttribute('download', fileName);
    }

    const href = URL.createObjectURL(blob);
    $a.href = href;
    $a.setAttribute('target', '_blank');
    $a.click();
    URL.revokeObjectURL(href);
}

async _post(url, body, signal) {
    const requestOptions = {
        method: "POST",
    }

    if (body) {
        requestOptions.body = body;

        // S'il s'agit d'un objet et non d'un formulaire
        if (!(body instanceof FormData)) {
            const headers = new Headers();
            headers.append("Content-Type", "application/json");

            requestOptions.headers = headers;
            requestOptions.body = JSON.stringify(body);;
        }
    }

    if (signal) {
        requestOptions.signal = signal;
    }

    return await fetch(url, requestOptions);
}
